## Phase 3 — Write the Smart Contract (`CertVerify.sol`)

**Goal**: Implement the core Solidity contract with all required data structures, functions, modifiers, and events.

### 3.1 Create `contracts/CertVerify.sol`

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.19;

contract CertVerify {

    // ─── State Variables ───────────────────────────────────────────────────────
    address public owner;
    mapping(address => bool) public issuers;
    mapping(bytes32 => Certificate) private certificates;

    // ─── Data Structures ───────────────────────────────────────────────────────
    struct Certificate {
        string  recipientName;
        bytes32 certificateHash;
        address issuerAddress;
        uint256 issuedDate;
        uint256 expiryDate;
        string  courseName;
        string  grade;
        bool    isActive;
    }

    // ─── Events ────────────────────────────────────────────────────────────────
    event CertificateIssued(
        bytes32 indexed certHash,
        address indexed issuer,
        string  recipientName,
        string  courseName
    );
    event CertificateRevoked(bytes32 indexed certHash, address indexed revokedBy);

    // ─── Modifiers ─────────────────────────────────────────────────────────────
    modifier onlyOwner() {
        require(msg.sender == owner, "Only owner can call this");
        _;
    }

    modifier onlyIssuer() {
        require(issuers[msg.sender], "Only authorized issuers can call this");
        _;
    }

    // ─── Constructor ───────────────────────────────────────────────────────────
    constructor() {
        owner = msg.sender;
        issuers[msg.sender] = true; // deployer is first issuer
    }

    // ─── Issuer Management ─────────────────────────────────────────────────────
    function addIssuer(address _issuer) public onlyOwner {
        require(_issuer != address(0), "Invalid address");
        issuers[_issuer] = true;
    }

    function removeIssuer(address _issuer) public onlyOwner {
        issuers[_issuer] = false;
    }

    // ─── Core Functions ────────────────────────────────────────────────────────

    /// @notice Issue a new certificate. Only authorized issuers can call this.
    function issueCertificate(
        string  memory _recipientName,
        string  memory _courseName,
        string  memory _grade,
        uint256        _expiryDate
    ) public onlyIssuer returns (bytes32) {
        require(bytes(_recipientName).length > 0, "Recipient name required");
        require(bytes(_courseName).length > 0,    "Course name required");
        require(_expiryDate > block.timestamp,    "Expiry must be in the future");

        // Generate a unique deterministic hash from certificate data + timestamp
        bytes32 certHash = keccak256(
            abi.encodePacked(
                _recipientName,
                _courseName,
                _grade,
                msg.sender,
                block.timestamp
            )
        );

        require(certificates[certHash].issuedDate == 0, "Certificate already exists");

        certificates[certHash] = Certificate({
            recipientName:   _recipientName,
            certificateHash: certHash,
            issuerAddress:   msg.sender,
            issuedDate:      block.timestamp,
            expiryDate:      _expiryDate,
            courseName:      _courseName,
            grade:           _grade,
            isActive:        true
        });

        emit CertificateIssued(certHash, msg.sender, _recipientName, _courseName);
        return certHash;
    }

    /// @notice Verify a certificate — returns validity status string.
    function verifyCertificate(bytes32 _certHash)
        public view
        returns (bool valid, string memory status, Certificate memory cert)
    {
        cert = certificates[_certHash];

        if (cert.issuedDate == 0) {
            return (false, "NOT_FOUND", cert);
        }
        if (!cert.isActive) {
            return (false, "REVOKED", cert);
        }
        if (block.timestamp > cert.expiryDate) {
            return (false, "EXPIRED", cert);
        }
        return (true, "VALID", cert);
    }

    /// @notice Retrieve raw certificate data.
    function getCertificate(bytes32 _certHash)
        public view
        returns (Certificate memory)
    {
        require(certificates[_certHash].issuedDate != 0, "Certificate not found");
        return certificates[_certHash];
    }

    /// @notice Revoke a certificate. Only issuers can revoke.
    function revokeCertificate(bytes32 _certHash) public onlyIssuer {
        require(certificates[_certHash].issuedDate != 0, "Certificate not found");
        require(certificates[_certHash].isActive,         "Already revoked");
        certificates[_certHash].isActive = false;
        emit CertificateRevoked(_certHash, msg.sender);
    }
}
```

### Key Solidity Concepts Used
| Concept | Where Used |
| :--- | :--- |
| `msg.sender` | Identifies who calls a function |
| `block.timestamp` | Records issue time; checks expiry |
| `require()` | Input validation & access control |
| `modifier onlyIssuer` | Restricts write functions to authorized addresses |
| `keccak256()` | Generates unique deterministic certificate hash |
| `struct Certificate` | Bundles all cert fields into one unit |
| `mapping(bytes32 => Certificate)` | O(1) lookup by certificate hash |
| `event CertificateIssued` | Logs issuance for off-chain indexing |

## Phase 7 — Web3.js Utility (`client/src/utils.js`)

**Goal**: Create the bridge layer between HTML frontend and the deployed smart contract.  
Pattern from `REFERENCE_KNOWLEDGE.md` — Pattern 2.

### 7.1 Create `client/src/utils.js`
```javascript
import Web3 from "../node_modules/web3/dist/web3.min.js";
import CertVerifyArtifact from "./contracts/CertVerify.json" assert { type: "json" };

/**
 * Step 1: Connect to Ganache RPC and create Web3 instance.
 */
export async function getWeb3() {
  const provider = new Web3.providers.HttpProvider("http://127.0.0.1:7545");
  const web3 = new Web3(provider);
  return web3;
}

/**
 * Step 2: Dynamically resolve deployed contract using network ID.
 * This approach (from truffleEng reference) always finds the right
 * contract address regardless of redeployments.
 */
export async function getContract(web3) {
  const networkId = await web3.eth.net.getId();
  const deployedNetwork = CertVerifyArtifact.networks[networkId];

  if (!deployedNetwork) {
    throw new Error(
      `CertVerify contract not deployed on network ${networkId}. ` +
      `Run: truffle migrate --network development`
    );
  }

  const contract = new web3.eth.Contract(
    CertVerifyArtifact.abi,
    deployedNetwork.address
  );

  return contract;
}

/**
 * Step 3: Get the active Ganache account (used as 'from' in .send() calls).
 */
export async function getCurrentAccount(web3) {
  const accounts = await web3.eth.getAccounts();
  if (!accounts || accounts.length === 0) {
    throw new Error("No Ganache accounts found. Is Ganache running?");
  }
  return accounts[0]; // accounts[0] = deployer = first authorized issuer
}
```

## Phase 9 — Frontend Logic (`client/src/index.js`)

**Goal**: Wire form submissions to smart contract calls using Web3.js.

### 9.1 Create `client/src/index.js`
```javascript
import { getWeb3, getContract, getCurrentAccount } from "./utils.js";

let web3, contract, account;

// ── Initialize on page load ──────────────────────────────────────────────────
window.addEventListener("load", async () => {
  try {
    web3     = await getWeb3();
    contract = await getContract(web3);
    account  = await getCurrentAccount(web3);

    document.getElementById("current-account").textContent = account;
    console.log("✅ Connected to CertVerify contract");
  } catch (err) {
    alert("❌ Could not connect to blockchain: " + err.message);
    console.error(err);
  }
});

// ── Helper: Show result in a result box ─────────────────────────────────────
function showResult(elementId, message, type = "success") {
  const el = document.getElementById(elementId);
  el.classList.remove("hidden", "success", "error", "warning");
  el.classList.add(type);
  el.innerHTML = message;
}

// ── Convert date string to UNIX timestamp ────────────────────────────────────
function dateToTimestamp(dateStr) {
  return Math.floor(new Date(dateStr).getTime() / 1000);
}

// ── ISSUE CERTIFICATE ────────────────────────────────────────────────────────
document.getElementById("issue-form").addEventListener("submit", async (e) => {
  e.preventDefault();

  const recipientName = document.getElementById("recipient-name").value.trim();
  const courseName    = document.getElementById("course-name").value.trim();
  const grade         = document.getElementById("grade").value.trim();
  const expiryDate    = document.getElementById("expiry-date").value;
  const expiryTs      = dateToTimestamp(expiryDate);

  showResult("issue-result", "⏳ Sending transaction to blockchain...", "info");

  try {
    // .send() — writes to blockchain, costs gas
    const receipt = await contract.methods
      .issueCertificate(recipientName, courseName, grade, expiryTs)
      .send({ from: account });

    // Extract the returned certHash from the event logs
    const certHash = receipt.events.CertificateIssued.returnValues.certHash;

    showResult("issue-result",
      `✅ <strong>Certificate Issued!</strong><br/>
       📋 Tx Hash: <code>${receipt.transactionHash}</code><br/>
       🔑 Cert Hash: <code>${certHash}</code><br/>
       <em>Save this hash — it is needed for verification.</em>`,
      "success"
    );
  } catch (err) {
    showResult("issue-result", `❌ Error: ${err.message}`, "error");
  }
});

// ── VERIFY CERTIFICATE ───────────────────────────────────────────────────────
document.getElementById("verify-form").addEventListener("submit", async (e) => {
  e.preventDefault();

  const certHash = document.getElementById("cert-hash").value.trim();

  try {
    // .call() — read-only, 0 gas fee
    const result = await contract.methods.verifyCertificate(certHash).call();
    const { valid, status, cert } = result;

    if (valid) {
      const issued = new Date(cert.issuedDate * 1000).toLocaleDateString();
      const expiry = new Date(cert.expiryDate * 1000).toLocaleDateString();
      showResult("verify-result",
        `✅ <strong>Certificate VALID</strong><br/>
         👤 Recipient: ${cert.recipientName}<br/>
         📚 Course: ${cert.courseName}<br/>
         🏅 Grade: ${cert.grade}<br/>
         📅 Issued: ${issued}<br/>
         ⏳ Expires: ${expiry}<br/>
         🏛️ Issuer: <code>${cert.issuerAddress}</code>`,
        "success"
      );
    } else if (status === "REVOKED") {
      showResult("verify-result", "❌ Certificate has been <strong>REVOKED</strong>.", "error");
    } else if (status === "EXPIRED") {
      showResult("verify-result", "⚠️ Certificate has <strong>EXPIRED</strong>.", "warning");
    } else {
      showResult("verify-result", "❌ Certificate <strong>NOT FOUND</strong> on blockchain.", "error");
    }
  } catch (err) {
    showResult("verify-result", `❌ Error: ${err.message}`, "error");
  }
});

// ── REVOKE CERTIFICATE ───────────────────────────────────────────────────────
document.getElementById("revoke-form").addEventListener("submit", async (e) => {
  e.preventDefault();

  const revokeHash = document.getElementById("revoke-hash").value.trim();

  showResult("revoke-result", "⏳ Sending revocation transaction...", "info");

  try {
    // .send() — writes to blockchain, costs gas
    const receipt = await contract.methods
      .revokeCertificate(revokeHash)
      .send({ from: account });

    showResult("revoke-result",
      `✅ <strong>Certificate Revoked!</strong><br/>
       📋 Tx Hash: <code>${receipt.transactionHash}</code>`,
      "success"
    );
  } catch (err) {
    showResult("revoke-result", `❌ Error: ${err.message}`, "error");
  }
});
```

## Common Errors & Fixes

| Error | Likely Cause | Fix |
| :--- | :--- | :--- |
| `Could not connect to blockchain` | Ganache not running | Start Ganache first |
| `Contract not deployed on network X` | Restarted Ganache (new network ID) | Re-run `truffle migrate --reset` |
| `Sender not authorized` | Calling as non-issuer account | Ensure `account[0]` is used |
| `Expiry must be in the future` | Past date selected | Select a future expiry date |
| `truffle: command not found` | Truffle not installed globally | `npm install -g truffle` |
| `CertVerify.json not found` | `contracts_build_directory` wrong | Verify `truffle-config.js` path |
